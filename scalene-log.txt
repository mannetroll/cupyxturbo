|   146 |        |       |       |            seed = (seed * IMM + IT) % ID                                                                             |
|   147 |        |       |       |            r = np.float32(seed) / np.float32(ID)                                                                     |
|   148 |        |       |       |                                                                                                                      |
|   149 |        |       |       |            # random_from_vec(r)                                                                                      |
|   150 |        |       |       |            idx = int(float(r) * 97.0)                                                                                |

|   154 |        |       |       |                idx = 96                                                                                              |
|   155 |        |       |       |            v = RANVEC[idx]                                                                                           |
|   156 |        |       |       |            RANVEC[idx] = r                                                                                           |
|   157 |        |       |       |                                                                                                                      |
|   158 |        |       |       |            th = np.float32(2.0) * PI * v                                                                             |

|   164 |        |       |       |                                                                                                                      |

|   166 |        |       |       |                # ALFA(X) == 0: purely u1 mode                                                                        |

|   173 |        |       |       |                denom = np.float32(1.0) + (gz * gz) / (ax * ax)                                                       |
|   174 |        |       |       |                ABSW2 = np.float32(np.exp(- (K / K0) * (K / K0)) / (denom * NORM))                                    |
